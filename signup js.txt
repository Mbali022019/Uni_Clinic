    (function() {
      // ============================================
      // PASSWORD CONSTRAINTS VALIDATION
      // ============================================
      
      const passwordInput = document.getElementById('password');
      const confirmPasswordInput = document.getElementById('confirm_password');
      
      function updatePasswordConstraints(password) {
        const length = password.length >= 8;
        const uppercase = /[A-Z]/.test(password);
        const number = /[0-9]/.test(password);
        const special = /[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password);
        
        // Update constraint indicators
        updateConstraint('lengthCheck', length);
        updateConstraint('uppercaseCheck', uppercase);
        updateConstraint('numberCheck', number);
        updateConstraint('specialCheck', special);
        
        // Update password input styling
        if (password.length > 0) {
          if (length && uppercase && number && special) {
            passwordInput.closest('.field').classList.remove('field-error');
            passwordInput.closest('.field').classList.add('field-success');
          } else {
            passwordInput.closest('.field').classList.remove('field-success');
            passwordInput.closest('.field').classList.add('field-error');
          }
        } else {
          passwordInput.closest('.field').classList.remove('field-error', 'field-success');
        }
        
        return { length, uppercase, number, special };
      }
      
      function updateConstraint(id, valid) {
        const element = document.getElementById(id);
        if (element) {
          if (valid) {
            element.classList.add('valid');
            element.querySelector('.pwd-constraint-icon').textContent = '✓';
          } else {
            element.classList.remove('valid');
            element.querySelector('.pwd-constraint-icon').textContent = '✕';
          }
        }
      }
      
      if (passwordInput) {
        passwordInput.addEventListener('input', function() {
          updatePasswordConstraints(this.value);
        });
      }

      // ============================================
      // REAL-TIME FIELD VALIDATION
      // ============================================
      
      function validateEmail(email) {
        const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        return emailPattern.test(email);
      }
      
      function validatePhone(phone) {
        const phonePattern = /^[0-9]{10}$/;
        return phone === '' || phonePattern.test(phone.replace(/\D/g, ''));
      }
      
      function setFieldValidation(fieldId, isValid) {
        const field = document.getElementById(fieldId);
        if (field) {
          const fieldWrapper = field.closest('.field');
          if (fieldWrapper) {
            if (field.value.trim() === '') {
              fieldWrapper.classList.remove('field-error', 'field-success');
            } else if (isValid) {
              fieldWrapper.classList.remove('field-error');
              fieldWrapper.classList.add('field-success');
            } else {
              fieldWrapper.classList.remove('field-success');
              fieldWrapper.classList.add('field-error');
            }
          }
        }
      }
      
      // Email validation
      const emailInput = document.getElementById('email');
      if (emailInput) {
        emailInput.addEventListener('input', function() {
          const isValid = this.value.trim() === '' || validateEmail(this.value.trim());
          setFieldValidation('email', isValid);
        });
      }
      
      // Phone validation
      const phoneInput = document.getElementById('phone');
      if (phoneInput) {
        phoneInput.addEventListener('input', function() {
          const isValid = this.value.trim() === '' || validatePhone(this.value.trim());
          setFieldValidation('phone', isValid);
        });
      }

          if (fieldWrapper) {
            if (this.value === '') {
              fieldWrapper.classList.remove('field-error', 'field-success');
            } else if (match) {
              fieldWrapper.classList.remove('field-error');
              fieldWrapper.classList.add('field-success');
            } else {
              fieldWrapper.classList.remove('field-success');
              fieldWrapper.classList.add('field-error');
            }
          }
        });
      }

          }

              eyeSvg.innerHTML = '<path d="M17.94 17.94A10.07 10.07 0 0 1 12 20c-7 0-11-8-11-8a18.45 18.45 0 0 1 5.06-5.94M9.9 4.24A9.12 9.12 0 0 1 12 4c7 0 11 8 11 8a18.5 18.5 0 0 1-2.16 3.19m-6.72-1.07a3 3 0 1 1-4.24-4.24"/><line x1="1" y1="1" x2="23" y2="23"/>';

          e.preventDefault();
          
          // Get all field values
          const fullName = document.getElementById('full_name')?.value.trim();
          const studentNum = document.getElementById('student_number')?.value.trim();
          const email = document.getElementById('email')?.value.trim();
          const phone = document.getElementById('phone')?.value.trim();
          const role = document.getElementById('roleSelect')?.value;
          const password = document.getElementById('password')?.value;
          const confirmPwd = document.getElementById('confirm_password')?.value;
          
          const roleMap = {
            patient: 'student',
            nurse: 'staff',
            admin: 'admin',
            student: 'student',
            staff: 'staff'
          };
          const dbRole = roleMap[role] || role;
          
          // Clear previous error
          if (msgDiv) msgDiv.innerText = '';
          
          // Validation checks
          if (!fullName || !studentNum || !email || !password || !confirmPwd) {
            if (msgDiv) msgDiv.innerText = 'Please fill in all required fields (*).';
            return;
          }
          
          if (!role || role === '') {
            if (msgDiv) msgDiv.innerText = 'Please select a role.';
            return;
          }
          
          if (!validateEmail(email)) {
            if (msgDiv) msgDiv.innerText = 'Please enter a valid email address.';
            return;
          }
          
          if (phone && !validatePhone(phone)) {
            if (msgDiv) msgDiv.innerText = 'Phone number must be exactly 10 digits.';
            return;
          }
          
          const pwdConstraints = updatePasswordConstraints(password);
          if (!pwdConstraints.length || !pwdConstraints.uppercase || !pwdConstraints.number || !pwdConstraints.special) {
            if (msgDiv) msgDiv.innerText = 'Password must have 8+ characters, uppercase, number, and special character.';
            return;
          }
          
          if (password !== confirmPwd) {
            if (msgDiv) msgDiv.innerText = 'Passwords do not match.';
            return;
          }
          
          // Check if supabase is available
          if (!window.supabaseClient) {
            if (msgDiv) {
              msgDiv.style.color = '#e53e3e';
              msgDiv.innerText = 'Error: Supabase client not configured.';
            }
            console.error('Supabase client not found');
            return;
          }
          
          // Disable button and show loading state
          submitBtn.disabled = true;
          submitBtn.innerText = 'Creating Account...';
          
          try {
            const supabase = window.supabaseClient;
            
            // Check if email or patient/staff number already exists
            const { data: existingUser, error: existingError } = await supabase
              .from('users')
              .select('email, student_number')
              .or(`email.eq.${email},student_number.eq.${studentNum}`)
              .maybeSingle();
            
            if (existingError) throw new Error('Account verification failed: ' + existingError.message);
            
            if (existingUser) {
              if (existingUser.email === email) {
                throw new Error('An account with this email already exists.');
              }
              if (existingUser.student_number === studentNum) {
                throw new Error('That patient/staff number is already registered.');
              }
              throw new Error('An account with this information already exists.');
            }
            
            // Get role ID
            const { data: roleData, error: roleError } = await supabase
              .from('roles')
              .select('role_id')
              .eq('role_name', dbRole)
              .single();
            
            if (roleError || !roleData) {
              throw new Error('Failed to retrieve role information.');
            }
            
            // Insert new user
            const { data, error: insertError } = await supabase
              .from('users')
              .insert([
                {
                  role_id: roleData.role_id,
                  student_number: studentNum,
                  full_name: fullName,
                  email: email,
                  password: password,
                  phone: phone || null,
                  patient_type: null
                }
              ])
              .select();
            
            if (insertError) {
              throw new Error('Account creation failed: ' + insertError.message);
            }
            
            // Success
            if (msgDiv) {
              msgDiv.style.color = '#16a34a';
              msgDiv.innerText = 'Account created successfully! Redirecting to login...';
            }
            
            console.log('User created:', data);
            form.reset();
            
            setTimeout(() => {
              window.location.href = 'login.html';
            }, 1500);
            
          } catch (err) {
            console.error('Signup error:', err);
            if (msgDiv) {
              msgDiv.style.color = '#e53e3e';
              msgDiv.innerText = err.message || 'An error occurred during signup. Please try again.';
            }
          } finally {
            submitBtn.disabled = false;
            submitBtn.innerText = 'Create Account';
          }
        });
      }
      
      // Sign in link behavior: redirect to login page
      const signInLink = document.getElementById('signInLink');
      if (signInLink) {
        signInLink.addEventListener('click', function(e) {
          e.preventDefault();
          window.location.href = 'login.html';
        });
      }
    })();
  </script>
