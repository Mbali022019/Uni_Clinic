  <!-- inline scripts for password toggle, form validation, and forgot password modal -->
  <script>
    (function() {
      // ============================================
      // PASSWORD TOGGLE
      // ============================================
      
      const toggleBtn = document.getElementById('togglePassword');
      const passwordField = document.getElementById('password');
      
      if (toggleBtn && passwordField) {
        toggleBtn.addEventListener('click', function(e) {
          e.preventDefault();
          const type = passwordField.getAttribute('type') === 'password' ? 'text' : 'password';
          passwordField.setAttribute('type', type);
          const eyeSvg = toggleBtn.querySelector('svg');
          if (eyeSvg) {
            if (type === 'text') {
              eyeSvg.innerHTML = '<path d="M17.94 17.94A10.07 10.07 0 0 1 12 20c-7 0-11-8-11-8a18.45 18.45 0 0 1 5.06-5.94M9.9 4.24A9.12 9.12 0 0 1 12 4c7 0 11 8 11 8a18.5 18.5 0 0 1-2.16 3.19m-6.72-1.07a3 3 0 1 1-4.24-4.24"/><line x1="1" y1="1" x2="23" y2="23"/>';
            } else {
              eyeSvg.innerHTML = '<path d="M1 12s4-8 11-8 11 8 11 8-4 8-11 8-11-8-11-8z"/><circle cx="12" cy="12" r="3"/>';
            }
          }
        });
      }

      // ============================================
      // LOGIN FORM SUBMISSION
      // ============================================
      
      const loginForm = document.getElementById('loginForm');
      const loginMessage = document.getElementById('loginMessage');
      const loginBtn = document.getElementById('loginBtn');
      
      if (loginForm) {
        loginForm.addEventListener('submit', async function(e) {
          e.preventDefault();
          
          const loginInput = document.getElementById('loginInput')?.value.trim();
          const password = document.getElementById('password')?.value;
          const role = document.getElementById('roleSelect')?.value;
          
          // Clear message
          if (loginMessage) loginMessage.innerText = '';
          
          // Validation
          if (!loginInput || !password || !role) {
            if (loginMessage) {
              loginMessage.classList.remove('success');
              loginMessage.innerText = 'Please fill in all fields.';
            }
            return;
          }
          
          loginBtn.disabled = true;
          loginBtn.innerText = 'Logging In...';
          
          try {
            // Check if supabase is available
            if (!window.supabaseClient) {
              throw new Error('Authentication service not configured. Please contact support.');
            }
            
            const supabase = window.supabaseClient;
            
            // Query users table for matching email or student number
            const { data, error } = await supabase
              .from('users')
              .select(`
                user_id,
                email,
                student_number,
                full_name,
                password,
                role_id,
                roles (
                  role_id,
                  role_name
                )
              `)
              .or(`email.eq.${loginInput},student_number.eq.${loginInput}`)
              .maybeSingle();
            
            if (error) {
              console.error('Query error:', error);
              throw new Error('Unable to verify account. Please try again.');
            }
            
            if (!data) {
              throw new Error('Account not found. Please check your email/patient number or create an account.');
            }
            
            // Verify password matches
            if (data.password !== password) {
              throw new Error('Invalid password. Please try again.');
            }
            
            // Verify role matches
            if (!data.roles || data.roles.role_name.toLowerCase() !== role.toLowerCase()) {
              throw new Error('Incorrect role selected for this account.');
            }
            
            // Store user data in localStorage
            localStorage.setItem('uniclinic_user', JSON.stringify(data));
            
            // Success
            if (loginMessage) {
              loginMessage.classList.add('success');
              loginMessage.innerText = `Login successful! Welcome ${data.full_name || 'back'}...`;
            }
            
            console.log('User logged in:', data);
            
            // Redirect based on role
            setTimeout(() => {
              const userRole = data.roles.role_name.toLowerCase();
              if (userRole === 'staff' || userRole === 'nurse') {
                window.location.href = 'nurseDash.html';
              } else if (userRole === 'admin') {
                window.location.href = 'adminDash.html';
              } else {
                window.location.href = 'sDashboard.html';
              }
            }, 1500);
            
          } catch (err) {
            console.error('Login error:', err);
            if (loginMessage) {
              loginMessage.classList.remove('success');
              loginMessage.innerText = err.message || 'An error occurred. Please try again.';
            }
          } finally {
            loginBtn.disabled = false;
            loginBtn.innerText = 'Log In';
          }
        });
      }

      // ============================================
      // FORGOT PASSWORD MODAL
      // ============================================
      
      const forgotPasswordBtn = document.getElementById('forgotPasswordBtn');
      const forgotPasswordModal = document.getElementById('forgotPasswordModal');
      const modalClose = document.getElementById('modalClose');
      const cancelReset = document.getElementById('cancelReset');
      const resetForm = document.getElementById('resetForm');
      const resetMessage = document.getElementById('resetMessage');
      const sendResetBtn = document.getElementById('sendResetBtn');
      
      // Open modal
      if (forgotPasswordBtn) {
        forgotPasswordBtn.addEventListener('click', function(e) {
          e.preventDefault();
          if (forgotPasswordModal) {
            forgotPasswordModal.classList.add('active');
            document.getElementById('resetEmail').value = '';
            if (resetMessage) resetMessage.innerText = '';
          }
        });
      }
      
      // Close modal
      const closeModal = () => {
        if (forgotPasswordModal) {
          forgotPasswordModal.classList.remove('active');
        }
      };
      
      if (modalClose) {
        modalClose.addEventListener('click', closeModal);
      }
      
      if (cancelReset) {
        cancelReset.addEventListener('click', closeModal);
      }
      
      // Close modal when clicking outside
      if (forgotPasswordModal) {
        forgotPasswordModal.addEventListener('click', function(e) {
          if (e.target === forgotPasswordModal) {
            closeModal();
          }
        });
      }
      
      // Reset form submission
      if (resetForm) {
        resetForm.addEventListener('submit', async function(e) {
          e.preventDefault();
          
          const email = document.getElementById('resetEmail')?.value.trim();
          
          if (!email) {
            if (resetMessage) {
              resetMessage.classList.remove('success');
              resetMessage.innerText = 'Please enter your email address.';
            }
            return;
          }
          
          sendResetBtn.disabled = true;
          sendResetBtn.innerText = 'Sending...';
          
          try {
            // Check if supabase is available
            if (!window.supabaseClient) {
              throw new Error('Service not available. Please try again later.');
            }
            
            const supabase = window.supabaseClient;
            
            // Send password reset email
            const { error } = await supabase.auth.resetPasswordForEmail(email, {
              redirectTo: window.location.origin + '/frontend/reset-password.html'
            });
            
            if (error) {
              throw new Error(error.message || 'Failed to send reset link.');
            }
            
            // Success
            if (resetMessage) {
              resetMessage.classList.add('success');
              resetMessage.innerText = 'Password reset link sent! Check your email.';
            }
            
            console.log('Password reset email sent to:', email);
            
            // Close modal after 2 seconds
            setTimeout(() => {
              closeModal();
            }, 2000);
            
          } catch (err) {
            console.error('Reset error:', err);
            if (resetMessage) {
              resetMessage.classList.remove('success');
              resetMessage.innerText = err.message || 'An error occurred. Please try again.';
            }
          } finally {
            sendResetBtn.disabled = false;
            sendResetBtn.innerText = 'Send Reset Link';
          }
        });
      }
    })();
  </script>
