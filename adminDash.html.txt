function escHtml(str) {
  return String(str)
    .replace(/&/g,'&amp;')
    .replace(/"/g,'&quot;')
    .replace(/</g,'&lt;')
    .replace(/>/g,'&gt;');
}

/* ═══════════════════════════════════════════
   SMART SEARCH — FIXED
   - Waits for cache to be populated
   - Correct field names (student_number vs user_id)
   - highlight works on null-safe values
   - navigates to correct page and scrolls to row
═══════════════════════════════════════════ */
function handleSmartSearch(query) {
  const results = document.getElementById('searchResults');
  query = (query || '').trim();

  if (query.length < 2) {
    results.innerHTML = '';
    results.style.display = 'none';
    return;
  }

  const q = query.toLowerCase();
  let html = '';

  /* ── Patients ── */
  const patientMatches = (window.searchCache.patients || []).filter(p =>
    (p.full_name || '').toLowerCase().includes(q) ||
    (p.student_number || '').toLowerCase().includes(q) ||
    (p.email || '').toLowerCase().includes(q)
  );

  if (patientMatches.length) {
    html += `<div class="search-section-title">Patients</div>`;
    patientMatches.slice(0, 4).forEach(p => {
      /* Use student_number as the data-id so rows are found correctly */
      const rowId = p.student_number || p.user_id || '';
      html += `
        <div class="search-result-item" onclick="searchGoTo('patients','${escHtml(String(rowId))}')">
          <div class="search-result-icon" style="background:var(--teal-100,#ccfbf1);">👤</div>
          <div>
            <div class="search-result-main">${highlightMatch(p.full_name || 'Unknown', query)}</div>
            <div class="search-result-sub">${escHtml(p.student_number || 'N/A')} · ${escHtml(p.email || '')}</div>
          </div>
        </div>`;
    });
  }

  /* ── Appointments ── */
  const apptMatches = (window.searchCache.appointments || []).filter(a =>
    (a.users?.full_name || '').toLowerCase().includes(q) ||
    (a.appointment_date || '').includes(q) ||
    (a.clinic_services?.service_name || '').toLowerCase().includes(q) ||
    (a.status || '').toLowerCase().includes(q)
  );

  if (apptMatches.length) {
    html += `<div class="search-section-title">Appointments</div>`;
    apptMatches.slice(0, 4).forEach(a => {
      const statusColor = a.status === 'Booked' ? 'var(--green-600)' : 'var(--slate-400)';
      html += `
        <div class="search-result-item" onclick="searchGoTo('appointments','${escHtml(String(a.appointment_id || ''))}')">
          <div class="search-result-icon" style="background:var(--blue-100,#dbeafe);">📅</div>
          <div>
            <div class="search-result-main">${highlightMatch(a.users?.full_name || 'Unknown', query)}</div>
            <div class="search-result-sub">
              ${escHtml(a.appointment_date || '')} · ${escHtml(a.clinic_services?.service_name || 'N/A')} ·
              <span style="color:${statusColor};">${escHtml(a.status || '')}</span>
            </div>
          </div>
        </div>`;
    });
  }

  /* ── Nurses ── */
  const nurseMatches = (window.searchCache.nurses || []).filter(n =>
    (n.full_name || '').toLowerCase().includes(q) ||
    (n.staff_profiles?.staff_number || '').toLowerCase().includes(q) ||
    (n.staff_profiles?.job_title || '').toLowerCase().includes(q)
  );

  if (nurseMatches.length) {
    html += `<div class="search-section-title">Nurses</div>`;
    nurseMatches.slice(0, 3).forEach(n => {
      html += `
        <div class="search-result-item" onclick="searchGoTo('nurses','${escHtml(String(n.user_id || ''))}')">
          <div class="search-result-icon" style="background:#f3e8ff;">👩‍⚕️</div>
          <div>
            <div class="search-result-main">${highlightMatch(n.full_name || 'Unknown', query)}</div>
            <div class="search-result-sub">${escHtml(n.staff_profiles?.staff_number || 'N/A')} · ${escHtml(n.staff_profiles?.job_title || 'Nurse')}</div>
          </div>
        </div>`;
    });
  }

  if (!html) {
    html = `<div class="search-empty">No results found for "${escHtml(query)}"</div>`;
  }

  results.innerHTML = html;
  results.style.display = 'block';
}

function highlightMatch(text, query) {
  if (!text || !query) return escHtml(text || '');
  /* escape special regex chars in query */
  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const regex = new RegExp('(' + escaped + ')', 'gi');
  return escHtml(text).replace(regex, '<span class="search-highlight">$1</span>');
}

function searchGoTo(page, id) {
  showPage(page, null);
  hideSearchResults();
  document.getElementById('smartSearch').value = '';

  /* Try to locate and highlight the matching row */
  setTimeout(() => {
    /* Rows use data-id attribute set during load */
    let row = document.querySelector(`[data-id="${CSS.escape(String(id))}"]`);

    /* Fallback: scan all rows for text match */
    if (!row) {
      const pageEl = document.getElementById('page-' + page);
      if (pageEl) {
        row = Array.from(pageEl.querySelectorAll('tbody tr')).find(r =>
          r.textContent.includes(id)
        );
      }
    }

    if (row) {
      row.scrollIntoView({ behavior: 'smooth', block: 'center' });
      row.style.transition = 'background 0.3s';
      row.style.background = '#fef08a';
      setTimeout(() => { row.style.background = ''; }, 2200);
    }
  }, 320);
}

async function sendChatMessage() {
  const input   = document.getElementById('chatInput');
  const message = (input.value || '').trim();
  if (!message) return;

  appendUserMessage(message);
  input.value = '';
  chatHistory.push({ role: 'user', content: message });
  showTypingIndicator();

  try {
    const response = await fetch('https://api.groq.com/openai/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${GROQ_API_KEY}`
      },
      body: JSON.stringify({
        model: 'llama-3.3-70b-versatile',
        messages: [
          { role: 'system', content: buildClinicContext() },
          ...chatHistory
        ],
        max_tokens: 500,
        temperature: 0.7
      })
    });

    const data  = await response.json();
    const reply = data.choices?.[0]?.message?.content ||
      'Sorry, I could not get a response. Please try again.';
    chatHistory.push({ role: 'assistant', content: reply });
    removeTypingIndicator();
    appendBotMessage(escHtml(reply).replace(/\n/g, '<br>'));

  } catch (err) {
    removeTypingIndicator();
    appendBotMessage('Sorry — I\'m having trouble connecting. Please try again.');
    console.error('Groq error:', err);
  }
}

function getStatusBadge(status) {
  switch (status) {
    case "Booked":     return "badge-yellow";
    case "Cancelled":  return "badge-red";
    case "Waitlisted": return "badge-blue";
    case "Completed":  return "badge-teal";
    default:           return "badge-teal";
  }
}

async function loadAI() {
  try {
    const predictions = await runPredictions();
    const { busiestDays, busiestHours, congestionRisk, highBookingDays } = predictions;

    /* Congestion banner */
    const banner = document.getElementById("aiCongestionBanner");
    if (banner) {
      document.getElementById("aiCongestionMessage").innerText = congestionRisk.message;
      if (congestionRisk.risk === "High") {
        banner.style.borderColor = "#dc2626";
        banner.style.background  = "#fef2f2";
      } else if (congestionRisk.risk === "Medium") {
        banner.style.borderColor = "#ea580c";
        banner.style.background  = "#fff7ed";
      }
    }

    /* Stat cards */
    safeSet("aiRiskLevel",   congestionRisk.risk);
    safeSet("aiBusiestDay",  busiestDays[0]?.day  || "--");
    safeSet("aiBusiestHour", busiestHours[0]?.hour || "--");
    safeSet("aiTodayCount",  congestionRisk.count);

    /* Busiest days chart */
    const maxDay = busiestDays[0]?.count || 1;
    const daysChartEl = document.getElementById("aiBusiestDaysChart");
    if (daysChartEl) {
      daysChartEl.innerHTML = busiestDays.map(d => `
        <div class="report-bar-row">
          <span class="report-bar-label">${escHtml(d.day)}</span>
          <div class="report-bar-track">
            <div class="report-bar-fill" style="width:${Math.round((d.count / maxDay) * 100)}%"></div>
          </div>
          <span class="report-bar-val">${d.count}</span>
        </div>`).join("");
    }

    /* Busiest hours chart */
    const maxHour = busiestHours[0]?.count || 1;
    const hoursChartEl = document.getElementById("aiBusiestHoursChart");
    if (hoursChartEl) {
      hoursChartEl.innerHTML = busiestHours.map(h => `
        <div class="report-bar-row">
          <span class="report-bar-label">${escHtml(h.hour)}</span>
          <div class="report-bar-track">
            <div class="report-bar-fill" style="width:${Math.round((h.count / maxHour) * 100)}%"></div>
          </div>
          <span class="report-bar-val">${h.count}</span>
        </div>`).join("");
    }

    /* High booking dates table */
    const maxCount = highBookingDays[0]?.count || 1;
    const bookingTableEl = document.getElementById("aiHighBookingTable");
    if (bookingTableEl) {
      bookingTableEl.innerHTML = highBookingDays.map(d => `
        <tr>
          <td>${escHtml(d.date)}</td>
          <td>${d.count}</td>
          <td>
            <div class="report-bar-track" style="width:100px;">
              <div class="report-bar-fill" style="width:${Math.round((d.count / maxCount) * 100)}%"></div>
            </div>
          </td>
        </tr>`).join("");
    }

    /* AI Recommendations */
    const recs = [];
    if (busiestDays[0])             recs.push(`📅 <strong>${escHtml(busiestDays[0].day)}</strong> is your busiest day — consider scheduling extra staff.`);
    if (busiestHours[0])            recs.push(`🕐 Peak hour is <strong>${escHtml(busiestHours[0].hour)}</strong> — ensure nurses are available then.`);
    if (congestionRisk.risk === "High")   recs.push(`🚨 High congestion risk today — activate walk-in overflow protocol.`);
    if (congestionRisk.risk === "Medium") recs.push(`⚠️ Moderate traffic today — monitor queue every 30 minutes.`);
    if (congestionRisk.risk === "Low")    recs.push(`✅ Low traffic today — good day for staff training or inventory checks.`);

    const recsEl = document.getElementById("aiRecommendations");
    if (recsEl) {
      recsEl.innerHTML = recs.map(r => `
        <div style="padding:12px;margin-bottom:10px;background:var(--slate-50);
          border-radius:var(--r-md);border-left:3px solid var(--teal-500);
          font-size:14px;line-height:1.5;">${r}</div>`).join("");
    }
  } catch (err) {
    console.error("loadAI error:", err);
  }

function escHtml(str) {
  return String(str)
    .replace(/&/g, "&amp;")
    .replace(/"/g, "&quot;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}
